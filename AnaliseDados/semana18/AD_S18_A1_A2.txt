Aula 1

// Nova tabela
calendario = 
ADDCOLUMNS(
    CALENDAR(DATE(2020, 1, 1), MAX(alunos_matriculados_por_turma[mes_referencia])),
    "Ano", YEAR([Date]),
    "Mês", FORMAT([Date], "MMMM"),
    "Trimestre", FORMAT([Date], "Q"),
    "Dia da Semana", FORMAT([Date], "w")
)

// Inserir na tabela alunos_matriculados_por_turma
mes_referencia = DATE([Ano],[Mês],1)

// Tabela de Medidas
Alunos Desmatriculados Ano Anterior = CALCULATE(
    SUM(alunos_matriculados_por_turma[Alunos Desmatriculados]),
    SAMEPERIODLASTYEAR('calendario'[Date])
)


Alunos Desmatriculados Mês Anterior = CALCULATE(
    SUM(alunos_matriculados_por_turma[Alunos Desmatriculados]),
    PREVIOUSMONTH('calendario'[Date])
)

Alunos Matriculados Ano Anterior = CALCULATE(
    SUM(alunos_matriculados_por_turma[Alunos Novos Matriculados]),
    SAMEPERIODLASTYEAR('calendario'[Date])
)
